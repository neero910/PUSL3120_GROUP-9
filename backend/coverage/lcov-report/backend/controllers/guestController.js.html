import Guest from '../models/Guest.js';
import mongoose from 'mongoose';
const byId = (id) => mongoose.isValidObjectId(id) ? { _id: id } : { _id: null };
export async function getAllGuests(req, res, next) { try { const { search } = req.query; const query = search ? { $or: [{ firstName: new RegExp(search, 'i') }, { lastName: new RegExp(search, 'i') }, { email: new RegExp(search, 'i') }, { phone: new RegExp(search, 'i') }] } : {}; const data = await Guest.find(query).sort({ createdAt: -1 }); return res.json({ success: true, count: data.length, data }); } catch (error) { return next(error); } }
export async function getGuestById(req, res, next) { try { const data = await Guest.findOne(byId(req.params.id)); if (!data) return res.status(404).json({ success: false, message: 'Guest not found' }); return res.json({ success: true, data }); } catch (error) { return next(error); } }
export async function createGuest(req, res, next) { try { const { firstName, lastName, email, phone } = req.body; if (!firstName || !lastName || !email || !phone) return res.status(400).json({ success: false, message: 'First name, last name, email, and phone are required' }); const data = await Guest.create({ ...req.body, email: email.toLowerCase(), nicPassport: req.body.nicPassport || req.body.identificationNumber }); return res.status(201).json({ success: true, message: 'Guest created successfully', data }); } catch (error) { return next(error); } }
export async function updateGuestData(req, res, next) { try { const data = await Guest.findOneAndUpdate(byId(req.params.id), req.body, { new: true, runValidators: true }); if (!data) return res.status(404).json({ success: false, message: 'Guest not found' }); return res.json({ success: true, message: 'Guest updated successfully', data }); } catch (error) { return next(error); } }
export async function deleteGuestData(req, res, next) { try { const data = await Guest.findOneAndDelete(byId(req.params.id)); if (!data) return res.status(404).json({ success: false, message: 'Guest not found' }); return res.json({ success: true, message: 'Guest deleted successfully', data }); } catch (error) { return next(error); } }
